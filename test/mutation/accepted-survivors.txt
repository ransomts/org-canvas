# Masked by the enclosing condition-case.  An empty vector is truthy in
# Elisp, so `or' takes the "use last folder" branch, `(elt [] -1)' signals,
# and the handler creates the folder anyway — the same observable outcome.
lisp/org-canvas-files.el :: and->or :: (if (and folders (> (length folders) 0))

# Guards error *logging* only.  What the function returns is decided by
# the cond below, so a spurious log line for a 2xx response changes
# nothing a test can observe.
lisp/org-canvas-files.el :: and->or :: (when (and status-line (>= (string-to-number status-line) 400))

# Equivalent.  Canvas ids are integers far below most-positive-fixnum
# (2305843009213693951), and `eq' compares fixnums by value.
lisp/org-canvas-files.el :: equal->eq :: (equal (alist-get 'folder_id f) folder-id))))

# `>' and `>=' differ only on ties — two folders whose paths are the same
# length — and either order is valid for deletion, since neither contains
# the other.
lisp/org-canvas-files.el :: gt->ge :: (> (length (or (alist-get 'full_name a) ""))

# Same masking as above: any length comparison still reaches the error
# path, which creates the folder.
lisp/org-canvas-files.el :: gt->ge :: (if (and folders (> (length folders) 0))
